21 '  let lightDistance = length(worldPos);'

23 '  let count = u32(u.occluderCount);'
24 '  for (var i: u32 = 0u; i < count; i = i + 1u) {'

26 '    let ocl = u.occluders[i];'
27 '    let toOcl = ocl.center - worldPos;'
28 '    let d = length(toOcl);'
29 '    if (d >= lightDistance) continue;              // sun itself (==) / past the light'
30 '    let toLightDir = normalize(-worldPos);'

33 '    let s = ocl.radius / max(d, 0.001);'
